cmake_minimum_required(VERSION 3.20)
project(BodySliderShadowlands LANGUAGES CXX)
enable_language(ASM_MASM)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Shadowlands requires Windows MSVC with -A x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()

# Reuse the numerical tests; WotLK profiles here are fixtures, not Shadowlands support.
add_executable(ShadowlandsShapeMathTest ../native/test_shape.cpp)
add_executable(ShadowlandsDeformMathTest ../native/test_deform.cpp)
foreach(target ShadowlandsShapeMathTest ShadowlandsDeformMathTest)
    target_compile_options(${target} PRIVATE /W4 /EHsc /UNDEBUG)
endforeach()
add_test(NAME shadowlands-shape-math-x64 COMMAND ShadowlandsShapeMathTest)
add_test(NAME shadowlands-deform-math-x64 COMMAND ShadowlandsDeformMathTest)

add_library(BodySliderShadowlands SHARED BodySliderShadowlands.cpp)
set_target_properties(BodySliderShadowlands PROPERTIES OUTPUT_NAME BodySliderShadowlandsRuntime-0.2.4)
target_compile_definitions(BodySliderShadowlands PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(BodySliderShadowlands PRIVATE /W4 /EHsc /permissive-)
target_link_libraries(BodySliderShadowlands PRIVATE bcrypt)
add_executable(ShadowlandsDllTest test_dll.cpp)
target_compile_options(ShadowlandsDllTest PRIVATE /W4 /EHsc /UNDEBUG)
add_test(NAME shadowlands-dll-rejects-unknown-host
         COMMAND ShadowlandsDllTest $<TARGET_FILE:BodySliderShadowlands>)
add_executable(ShadowlandsSizeTest test_size.cpp)
target_compile_options(ShadowlandsSizeTest PRIVATE /W4 /UNDEBUG)
add_test(NAME shadowlands-size-baseline-and-reset COMMAND ShadowlandsSizeTest)
add_executable(ShadowlandsTransformTest test_transform.cpp)
target_compile_options(ShadowlandsTransformTest PRIVATE /W4 /UNDEBUG)
add_test(NAME shadowlands-spatial-transform COMMAND ShadowlandsTransformTest)
add_library(ShadowlandsBridgeSelfTest SHARED test_bridge_dll.cpp)
target_compile_definitions(ShadowlandsBridgeSelfTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ShadowlandsBridgeSelfTest PRIVATE /W4 /EHsc)
target_link_libraries(ShadowlandsBridgeSelfTest PRIVATE bcrypt)

add_executable(BodySliderShadowlandsAttach Attach.cpp)
target_compile_definitions(BodySliderShadowlandsAttach PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(BodySliderShadowlandsAttach PRIVATE /W4 /EHsc /permissive-)
target_link_libraries(BodySliderShadowlandsAttach PRIVATE bcrypt)

add_library(ShadowlandsTlsLoader SHARED tls_loader.asm)
set_target_properties(ShadowlandsTlsLoader PROPERTIES LINKER_LANGUAGE CXX)
target_link_options(ShadowlandsTlsLoader PRIVATE /NOENTRY /NODEFAULTLIB /INCREMENTAL:NO /EXPORT:TlsLoader)
